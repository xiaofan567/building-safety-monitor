# 05 · 工具集

> 所有 Agent 工具统一走 Spring Boot REST API 转发，避免直接连接数据库。
> 这一章是 **"工具层 ↔ Spring Boot 端点"** 的映射表 + 实现规范。

## 1. 设计原则

1. **统一后端**：所有数据查询/写操作都走 Spring Boot REST，不直接连 MySQL
2. **统一鉴权**：Agent 调用后端时携带 `Admin-Token`（后端服务间调用的内部 token）
3. **统一格式**：所有工具返回 `{"ok": bool, "data": ..., "error": str|None}`
4. **统一护栏**：所有工具走 `safe_tool` 装饰器
5. **统一观测**：每次调用都生成 `tool` span

```python
@safe_tool(risk_level="low", max_calls_per_session=20, timeout=10)
async def knowledge_search(query: str, top_k: int = 5,
                          filter_category: Optional[str] = None) -> dict:
    """在内部知识库中检索文档片段。

    用于回答业务概念、API 用法、阈值定义、设备规格等问题。

    Args:
        query: 自然语言查询
        top_k: 返回片段数量（1-20）
        filter_category: 可选过滤 doc category（api/arch/database/...）

    Returns:
        {
          "chunks": [
            {"doc_id": "...", "content": "...", "score": 0.87,
             "metadata": {"category": "...", "source": "..."}}
          ],
          "summary": "..."   # 摘要
        }
    """
    result = await hybrid_retriever.search(
        query=query, top_k=min(top_k, 20), filters={"category": filter_category} if filter_category else None
    )
    return {
        "chunks": [{"doc_id": c["doc_id"], "content": c["content"],
                    "score": c["score"], "metadata": c["metadata"]} for c in result],
        "summary": summarize(result),
    }
```

## 3. BackendClient 实现

```python
# app/backend_client.py
import httpx
from app.config import settings

class BackendClient:
    """统一的 Spring Boot REST 调用客户端"""

    def __init__(self):
        self.base_url = settings.BACKEND_BASE_URL  # http://localhost:8080
        self.token = settings.BACKEND_ADMIN_TOKEN  # 内部服务间调用 token
        self.client = httpx.AsyncClient(
            base_url=self.base_url,
            headers={"Admin-Token": self.token,
                     "User-Agent": "agent-service/1.0"},
            timeout=httpx.Timeout(30.0, connect=5.0),
            limits=httpx.Limits(max_connections=50, max_keepalive_connections=20),
        )

    async def get(self, path: str, params: dict = None) -> dict:
        r = await self.client.get(path, params=params)
        r.raise_for_status()
        return r.json()

    async def post(self, path: str, json: dict = None, params: dict = None) -> dict:
        r = await self.client.post(path, json=json, params=params)
        r.raise_for_status()
        return r.json()

    async def close(self):
        await self.client.aclose()

backend_client = BackendClient()
```

### 4.1 鉴权

```yaml

agent:
  service:
    url: http://localhost:7000
    internal-token: ${AGENT_INTERNAL_TOKEN:agent-internal-secret}
  backend:
    internal-token: ${BACKEND_INTERNAL_TOKEN:backend-internal-secret}
```

```java
// AgentAuthFilter.java
@Component
@Order(Ordered.HIGHEST_PRECEDENCE + 10)
public class AgentAuthFilter implements WebFilter {
    @Value("${agent.backend.internal-token}")
    private String internalToken;

    @Override
    public Mono<Void> filter(ServerWebExchange exchange, WebFilterChain chain) {
        String path = exchange.getRequest().getPath().value();
        if (!path.startsWith("/agent/")) {
            return chain.filter(exchange);
        }
        // /agent/health 放行
        if (path.equals("/agent/health")) {
            return chain.filter(exchange);
        }
        String token = exchange.getRequest().getHeaders().getFirst("X-Internal-Token");
        if (!Objects.equals(token, internalToken)) {
            return Mono.error(new ResponseStatusException(HttpStatus.UNAUTHORIZED));
        }
        return chain.filter(exchange);
    }
}
```

```java
// AgentController.java
@RestController
@RequestMapping("/agent")
public class AgentController {
    @Autowired private AgentProxyService proxy;
    @Autowired private AgentSessionService sessionService;
    @Autowired private TimeTravelService timeTravel;

    @GetMapping("/health")
    public Map<String, Object> health() {
        return Map.of("status", "up", "service", "agent-bff");
    }

    @PostMapping(value = "/chat", produces = MediaType.TEXT_EVENT_STREAM_VALUE)
    public Flux<ServerSentEvent<String>> chat(@RequestBody ChatRequest req,
                                              @RequestHeader("Admin-Token") String token) {
        return proxy.streamChat(req, token);
    }

    @PostMapping("/chat/resume")
    public Flux<ServerSentEvent<String>> resume(@RequestBody ResumeRequest req) {
        return proxy.resume(req);
    }

    @GetMapping("/session/list")
    public List<SessionDto> listSessions(@RequestHeader("Admin-Token") String token) {
        return sessionService.listByUser(parseUserId(token));
    }

    @GetMapping("/session/{id}/checkpoints")
    public List<CheckpointDto> listCheckpoints(@PathVariable String id) {
        return timeTravel.listCheckpoints(id);
    }

    @PostMapping("/session/{id}/fork")
    public Map<String, String> fork(@PathVariable String id, @RequestBody ForkRequest req) {
        return Map.of("new_session_id", timeTravel.fork(id, req.checkpointId()));
    }

    @PostMapping(value = "/session/{id}/replay", produces = MediaType.TEXT_EVENT_STREAM_VALUE)
    public Flux<ServerSentEvent<String>> replay(@PathVariable String id,
                                                @RequestBody ReplayRequest req) {
        return timeTravel.replay(id, req);
    }
}
```

### 4.4 AgentProxyService（SSE 转发核心）

```java
// AgentProxyService.java
@Service
@Slf4j
public class AgentProxyService {
    @Autowired private WebClient agentWebClient;

    public Flux<ServerSentEvent<String>> streamChat(ChatRequest req, String userToken) {
        return agentWebClient.post()
                .uri("/agent/chat")
                .header("X-User-Token", userToken)
                .bodyValue(req)
                .retrieve()
                .bodyToFlux(String.class)
                .map(line -> ServerSentEvent.builder(line).build())
                .timeout(Duration.ofSeconds(120))
                .doOnError(e -> log.error("agent_chat_error", e));
    }
}
```

```python
# app/tools/equipment.py
from app.tools.decorators import safe_tool
from app.backend_client import backend_client

@safe_tool(risk_level="low", timeout=15)
async def query_equipment_list(state: AgentState,
                                asset_name: str = None,
                                type_id: int = None,
                                page_num: int = 1,
                                page_size: int = 20) -> dict:
    """查询设备资产列表。

    Args:
        asset_name: 设备名称（模糊匹配）
        type_id: 设备类型 ID
        page_num: 页码
        page_size: 每页条数

    Returns:
        {ok: True, data: {records: [...], total: n}}
    """
    body = {"pageNum": page_num, "pageSize": page_size}
    if asset_name:
        body["assetName"] = asset_name
    if type_id:
        body["typeId"] = type_id

    return {"ok": True, "data": await backend_client.post(
        "/equipmentAssets/list", json=body
    )}
```

### 5.3 调用外部 AI 服务示例

```python
# app/tools/ai_risk.py
@safe_tool(risk_level="low", timeout=30)
async def predict_tower_safety(state: AgentState, asset_id: int) -> dict:
    """调用 AIRecognize 的塔吊安全预测 LSTM 模型。

    Args:
        asset_id: 塔吊设备 ID

    Returns:
        {ok: True, data: {risk_level: "low|medium|high", score: 0.0-1.0, features: {...}}}
    """
    # 1. 先获取塔吊最近数据
    tower_data = await backend_client.post(
        f"/tower/getData/{asset_id}",
        json={"recent": True}
    )
    # 2. 调 AIRecognize
    async with httpx.AsyncClient() as client:
        r = await client.post(
            f"{settings.AI_RECOGNIZE_URL}/crane_safety",
            json={"tower_id": asset_id, "data": tower_data},
            timeout=20.0,
        )
        r.raise_for_status()
        return {"ok": True, "data": r.json()}
```
