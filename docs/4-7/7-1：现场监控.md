**难点**：HLS播放器管理、动态视频引用处理、全屏状态控制、组件间方法调用

**内容**：实现完整的视频监控查看系统，包括摄像头列表管理、多路视频播放、动态布局适配和全屏控制功能。

**2.2 HLS播放器初始化和管理**

```stylus
javascript
// 初始化单个HLS播放器
const initHlsPlayer = (videoElement, sourceUrl, index) => {
  if (!videoElement) return

  if (Hls.isSupported()) {
    // 创建HLS实例
    hlsInstances[index] = new Hls({
      debug: false,
      enableWorker: true,
      lowLatencyMode: true,
      backBufferLength: 90
    })

    // 加载视频源
    hlsInstances[index].loadSource(sourceUrl)
    hlsInstances[index].attachMedia(videoElement)

    // 事件监听
    hlsInstances[index].on(Hls.Events.MANIFEST_PARSED, () => {
      console.log(`摄像头 ${index + 1} M3U8解析成功`)
      videoElement.play().catch(e => {
        console.log(`摄像头 ${index + 1} 自动播放失败`, e)
      })
    })

    hlsInstances[index].on(Hls.Events.ERROR, (event, data) => {
      console.error(`摄像头 ${index + 1} HLS播放错误:`, data)
      // 错误恢复机制
      if (data.fatal) {
        switch (data.type) {
          case Hls.ErrorTypes.NETWORK_ERROR:
            setTimeout(() => hlsInstances[index].startLoad(), 3000)
            break
          case Hls.ErrorTypes.MEDIA_ERROR:
            hlsInstances[index].recoverMediaError()
            break
        }
      }
    })
  } else if (videoElement.canPlayType('application/vnd.apple.mpegurl')) {
    // Safari等原生支持HLS的浏览器
    videoElement.src = sourceUrl
    videoElement.addEventListener('canplay', () => {
      videoElement.play().catch(e => {
        console.log(`摄像头 ${index + 1} 自动播放失败`, e)
      })
    })
  }
}
```

**实现要点**：

检查浏览器对HLS的支持情况

配置HLS播放器参数优化性能

实现M3U8清单解析成功的自动播放

添加网络错误和媒体错误的恢复机制

兼容Safari等原生支持HLS的浏览器

          case Hls.ErrorTypes.MEDIA_ERROR:
            hlsInstances[index].recoverMediaError()
            break
        }
      }
    })

  } else if (videoElement.canPlayType('application/vnd.apple.mpegurl')) {
    videoElement.src = sourceUrl
    videoElement.addEventListener('canplay', () => {
      videoElement.play().catch(e => {
        console.log(`摄像头 ${index + 1} 自动播放失败`, e)
      })
    })
  }
}

  }
};
