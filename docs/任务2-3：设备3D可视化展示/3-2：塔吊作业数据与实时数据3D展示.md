任务指导

在工业设备管理系统中，操作人员在修改塔吊等重型设备的工作参数时，往往无法直观地预见到参数调整后设备的实际运行状态。这种"盲操作"模式存在以下问题：

- 无法验证参数设置的合理性
- 难以预测设备运行的安全边界
- 操作培训效果有限，缺乏直观反馈

在之前的任务中，我们已经完成了以下基础工作：

- **Three.js模型加载与动画实现** - 学习了如何使用Three.js加载3D模型并实现基础动画效果
- **设备TCP连接建立** - 实现了设备与监控系统的网络通信连接
- **数据模拟平台开发** - 构建了塔吊设备数据的模拟和传输平台

在本任务中，我们将整合前期成果，实现**塔吊参数设置与3D模型动态联动的可视化系统**。具体目标包括：

- 在参数设置弹窗中集成3D模型展示区域
- 实现参数调整时模型的实时动态响应

具体实现思路为通过事件监听和数据绑定，将参数设置界面与3D模型动画系统连接起来。

- 页面加载时预加载3D场景所需资源（场景、相机、渲染器、着色器、模型文件）
- 监听塔吊参数设置弹窗的打开事件，当弹窗打开时，初始化或显示3D模型展示区域
- 当参数调整时，通过事件机制更新3D模型状态

### 2．3D模型动画效果展示

上一小节功能仅实现了塔吊3D模型展示，但是还需实现模型动画效果和数据联动。在修改塔吊作业信息时，为直观展示作业情况，当修改塔吊参数时，右侧塔吊模型需模拟作业过程，从起始的角度、高度与半径触发，移动到结束角度、高度和半径。动画时间可设置为10秒钟。

为实现以上效果，可先定义起始与最终的角度。高度与半径数据。然后定义当前动画帧数。固定总动画帧数为600帧（10秒*屏幕默认60hz刷新率）。通过当前动画帧数计算当前的各个参数值，可参考代码如下。

```kotlin
    // 动画循环
    animate = () => {
        //当播放状态为True时才运行动画
        if (this.showTC) {
            requestAnimationFrame(this.animate);
        }
        //600帧为结束帧，大于结束帧则不再播放
        if (this.animationFrame <= 600) {
            //计算当前帧数的角度、塔车高度、塔车半径
            let targetAngle = this.startAngleAvg + this.animationFrame * ((this.endAngleAvg - this.startAngleAvg) / 600);
            let targetPointRadius = this.liftingPointRadiusAvg + this.animationFrame * ((this.liftingPointRadiusUnloadAvg - this.liftingPointRadiusAvg) / 600);
            let targetPointHeight = this.liftingPointHeightAvg + this.animationFrame * ((this.liftingPointHeightUnloadAvg - this.liftingPointHeightAvg) / 600);
            //更新塔车旋转角度、塔车位置,该方法可参考上一任务的代码
            this.updateRotationAngle(degreesToRadians(targetAngle))
            //更新塔车位置，同时传入半径和吊钩高度
            this.updateTrolleyPosition(mapTrolleyPosition(targetPointRadius), mapHookHeight(targetPointHeight))
            //播放下一帧
            this.animationFrame++;
        }



        this.controls.update();//更新控制器，如果部在循环中调用，则无法实现阻尼效果
        this.renderer.render(this.scene, this.camera);
    }
```

其中updateRotationAngle和updateTrolleyPosition为动画函数，可以通过此函数旋转塔吊、移动小车和吊钩，可参考上一任务。degreesToRadians、mapTrolleyPosition和mapHookHeight为数据映射函数，将实际的塔吊数据转换为3D模型坐标，分别为角度转弧度（degreesToRadians）、前臂小车映射函数（mapTrolleyPosition）和吊钩映射函数（mapHookHeight）。其中前臂小车映射函数将0~50米映射到-3400~3400坐标范围；吊钩映射函数将0~60米映射到-4600~1600范围。以上的具体值可通过上一任务中修改塔吊模型移动参数测试获取。

// 数据映射函数：将输入值从一个范围映射到另一个范围
function mapValue(value, inMin, inMax, outMin, outMax) {
    return ((value - inMin) * (outMax - outMin)) / (inMax - inMin) + outMin;
}

// 数据映射函数：根据高度值计算吊钩的Y轴位置
function heightToYPosition(height, minHeight = 0, maxHeight = 100) {
    // 假设在3D场景中，吊钩的高度范围对应Y轴的特定范围
    // 这里假设最小高度对应Y=0，最大高度对应Y=30
    return mapValue(height, minHeight, maxHeight, 0, 30);
}

/**
 * 将角度值转换为弧度制
 * @param {number} degrees 角度值
 * @returns {number} 弧度值
 */
function degreesToRadians(degrees) {
    return degrees * (Math.PI / 180);
}

/**
 * 将0~50范围的值映射到-3400~3400范围（前臂小车映射函数）
 * @param {number} value 0~50范围内的值
 * @returns {number} -3400~3400范围内的映射值
 */
function mapTrolleyPosition(value) {
    // 确保输入值在有效范围内
    const clampedValue = Math.max(0, Math.min(50, value));
    // 线性映射公式: newValue = (value - oldMin) * (newMax - newMin) / (oldMax - oldMin) + newMin
    return (clampedValue - 0) * (3400 - (-3400)) / (50 - 0) + (-3400);
}

/**
 * 将0~60范围的值映射到-4600~1600范围
 * @param {number} value 0~60范围内的值
 * @returns {number} -4600~1600范围内的映射值
 */
function mapHookHeight(value) {
    // 确保输入值在有效范围内
    const clampedValue = Math.max(0, Math.min(60, value));
    // 线性映射公式: newValue = (value - oldMin) * (newMax - newMin) / (oldMax - oldMin) + newMin
    return (clampedValue - 0) * (1600 - (-4600)) / (60 - 0) + (-4600);
}

    // 更新旋转角度
    updateRotationAngle(targetRadians) {
        if (!this.plane002) return;
        const delta = targetRadians - this.plane002.rotation.y;

        // 旋转塔吊主部件
        //先获取塔体的中心点与修正点之间的向量
        const offset = new THREE.Vector3().subVectors(this.plane002.position, this.rotationCenter);
        // 使这个向量沿着y轴执行旋转操作，旋转的角度为delta
        offset.applyAxisAngle(new THREE.Vector3(0, 1, 0), delta);
        //让塔体旋转
        this.plane002.rotation.y = targetRadians;
        //先设置塔体的位置为修正点，然后再加上偏移向量回到正常位置
        this.plane002.position.copy(this.rotationCenter).add(offset);
    }
    // 更新小车位置
    updateTrolleyPosition(trolleyPosition, hookHeight) {
        console.log('更新小车位置:', trolleyPosition, hookHeight);
        if (!this.trolley) return;
        // 设置小车位置
        this.trolley.position.set(trolleyPosition, 0, 0);
        this.updateHookAndRope(hookHeight)
    }
    // 更新吊钩和模型自带吊线
    updateHookAndRope(hookHeight) {
        if (!this.hook || !this.trolley || !this.rope) return;

        // 更新吊钩位置（保持在小车正下方）
        this.hook.position.x = this.trolley.position.x;
        this.hook.position.y = hookHeight + 100;//+100高度防止穿模
        //获取吊绳长度
        const box = new THREE.Box3().setFromObject(this.rope);
        let originalHeight = box.max.y - box.min.y; // 初始高度
        //更新高度缩放比例
        this.rope.scale.y = (1 - hookHeight / 1600);
        //同步更新吊绳位置
        this.rope.position.y = originalHeight * (1 - this.rope.scale.y) / 2;

        this.rope.position.x = this.trolley.position.x;
    }

    animate = () => {

        if (this.showTC) {
            requestAnimationFrame(this.animate);
        }
        //600帧为结束帧，大于结束帧则不再播放
        if (this.animationFrame <= 600) {
            //计算当前帧数的角度、塔车高度、塔车半径
            let targetAngle = this.startAngleAvg + this.animationFrame * ((this.endAngleAvg - this.startAngleAvg) / 600);
            let targetPointRadius = this.liftingPointRadiusAvg + this.animationFrame * ((this.liftingPointRadiusUnloadAvg - this.liftingPointRadiusAvg) / 600);
            let targetPointHeight = this.liftingPointHeightAvg + this.animationFrame * ((this.liftingPointHeightUnloadAvg - this.liftingPointHeightAvg) / 600);
            //更新塔车旋转角度、塔车位置,该方法可参考上一任务的代码
            this.updateRotationAngle(degreesToRadians(targetAngle))
            //更新塔车位置，同时传入半径和吊钩高度
            this.updateTrolleyPosition(mapTrolleyPosition(targetPointRadius), mapHookHeight(targetPointHeight))
            //播放下一帧
            this.animationFrame++;
        }



        this.controls.update();//更新控制器，如果部在循环中调用，则无法实现阻尼效果
        this.renderer.render(this.scene, this.camera);
    }

    async saveDeviceConfig() {
        if (!this.currentEditingDevice) return;

        const configKey = this.getConfigKey(this.currentEditingDevice);
        if (!configKey) return;

        const config = { ...this.deviceConfig[configKey] };
        let hasChanges = false;

        // 处理code和id
        const codeInput = document.getElementById('config_code');
        const idInput = document.getElementById('config_id');

        if (codeInput && codeInput.value !== config.code) {
            config.code = codeInput.value;
            hasChanges = true;
        }

        if (idInput) {
            // 处理id可能是数字或字符串
            const newId = isNaN(idInput.value) ? idInput.value : Number(idInput.value);
            if (newId !== config.id) {
                config.id = newId;
                hasChanges = true;
            }
        }

        // 处理其他参数
        Object.keys(config).forEach(key => {
            if (key !== 'code' && key !== 'id' && typeof config[key] === 'object') {
                const minInput = document.getElementById(`min_${key}`);
                const maxInput = document.getElementById(`max_${key}`);

                if (minInput && maxInput) {
                    const newMin = parseFloat(minInput.value);
                    const newMax = parseFloat(maxInput.value);

                    if (!isNaN(newMin) && !isNaN(newMax) &&
                        (newMin !== config[key].min || newMax !== config[key].max)) {
                        config[key] = { min: newMin, max: newMax };
                        hasChanges = true;
                    }
                }
            }
        });

        if (hasChanges) {
            this.deviceConfig[configKey] = config;

            try {
                const response = await fetch('/api/config', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify(this.deviceConfig),
                });

                const result = await response.json();
                if (result.success) {
                    this.addLogEntry('success', `设备 ${this.currentEditingDevice} 配置已保存`);
                    this.closeModals();
                } else {
                    this.addLogEntry('error', '保存配置失败');
                }
            } catch (error) {
                this.addLogEntry('error', '保存配置时发生错误');
            }
        } else {
            this.closeModals();
        }
    }

    closeModals() {

        this.showTC = false;
        this.currentEditingDevice = null;

    }

    addLogEntry(type, message, timestamp = new Date().toISOString()) {
        //获取日志框
        const logContainer = document.getElementById('logContent');
        // 添加一个日志节点
        const logEntry = document.createElement('div');
        // 设置日志class样式
        logEntry.className = `log-entry ${type} show`;
        // 设置日志节点显示内容
        const time = new Date(timestamp).toLocaleTimeString();
        logEntry.innerHTML = `
            <span class="timestamp">[${time}]</span>
            <span class="message">${message}</span>
        `;
        // 在日志框中添加日志节点
        logContainer.appendChild(logEntry);
        // 如果设置自动滚动，则滚动到最底部
        if (this.autoScroll) {
            logContainer.scrollTop = logContainer.scrollHeight;
        }

        this.filterLogs(); // 应用当前筛选
    }

    filterLogs() {
        const logEntries = document.querySelectorAll('.log-entry');
        logEntries.forEach(entry => {
            const isSent = entry.classList.contains('sent');
            const isReceived = entry.classList.contains('received');
            const isError = entry.classList.contains('error');
            const isInfo = entry.classList.contains('info');

            let shouldShow = false;

            switch (this.logFilter) {
                case 'all':
                    shouldShow = true;
                    break;
                case 'sent':
                    shouldShow = isSent || isInfo;
                    break;
                case 'received':
                    shouldShow = isReceived || isError;
                    break;
            }

            if (shouldShow) {
                entry.classList.add('show');
            } else {
                entry.classList.remove('show');
            }
        });
    }

// 递归查找部件
function findNestedPart(node, targetName, callback) {
    if (node.name === targetName) {
        callback(node);
        return true;
    }
    for (let child of node.children) {
        if (findNestedPart(child, targetName, callback)) return true;
    }
    return false;
}
