任务描述

本任务需要实现一个基于WebSocket通信的工程监测设备数据展示系统前端界面。通过HTML、CSS和JavaScript与Express后端服务建立Socket.IO通信，实时展示设备状态、连接状态和数据传输日志，并提供设备控制功能。

知识点：WebSocket通信原理、Socket实时数据通信服务、前端DOM操作、事件处理、异步编程、JSON数据解析；

**重点：** Socket.IO客户端连接与事件监听、动态UI更新、设备状态管理、实时数据展示；

**难点：** 多设备状态同步管理、实时日志过滤与展示、模态框交互逻辑、配置数据的动态加载与保存；

**内容：**完成设备通信模拟的操作界面展示和设备模拟控制。

可编写函数实现根据连接信息渲染连接状态按钮，参考代码如下：

```javascript
    updateConnectionStatus(status) {
        //获取连接状态框的DOM对象
        const container = document.getElementById('connectionStatus');
        //根据连接状态数据创建节点
        Object.entries(status).forEach(([group, connected]) => {
            const groupName = this.getGroupDisplayName(group);
            const elementId = `connection-${group}`;
            // 创建节点
            let element = document.getElementById(elementId);
            if (!element) {
                element = document.createElement('div');
                element.id = elementId;
                element.className = 'connection-item';
                container.appendChild(element);
            }
            // 根据连接状态修改样式
            element.className = `connection-item ${connected ? 'connected' : 'disconnected'}`;
            // 如果是已连接，需要添加断开连接按钮
            element.innerHTML = `
                <div>${groupName}</div>
                <img src="img/${connected?"connect.svg":"disconnect.svg"}"  class="disconnect-icon">
                ${connected ? `<button class="disconnect-btn" data-group="${group}">断开连接</button>` : ''}
            `;

            // 绑定断开连接事件
            if (connected) {
                element.querySelector('.disconnect-btn').addEventListener('click', (e) => {
                    e.stopPropagation();
                    this.showDisconnectConfirm(group, groupName);
                });
            }
        });
    }
```

```kotlin
        // 先判断当前是否选中设备组
        if (this.pendingDisconnectGroup) {
            // 向着服务器发送断开连接请求
            this.socket.emit('disconnect_group', { group: this.pendingDisconnectGroup });
            // 在日志中添加一条日志信息，该功能可后续实现
```

```xquery
    generateConfigForm(deviceId) {
        // 根据id获取是哪个设备配置
        const configKey = this.getConfigKey(deviceId);
        if (!configKey || !this.deviceConfig[configKey]) {
            return '<p>该设备暂无配置选项</p>';
        }
        const config = this.deviceConfig[configKey];
        // 表单头添加
        let html = '<div class="config-form">';
        // 表达基础内容添加，所有的设备都一样
        // 添加基本信息部分
        html += `
            <div class="config-section">
                <div class="config-section-title">基本信息</div>
                <div class="config-group double-width">
                    <label>设备编码 (code)</label>
                    <div class="config-inputs full-width">
                        <input type="text" id="config_code" placeholder="设备编码" value="${config.code || ''}">
                    </div>
                </div>
                <div class="config-group">
                    <label>设备ID (id)</label>
                    <div class="config-inputs full-width">
                        <input type="text" id="config_id" placeholder="设备ID" value="${config.id || ''}">
                    </div>
                </div>
            </div>
        `;

        // 根据config抽取出可配置的参数
        const hasParameters = Object.keys(config).some(key => 
            key !== 'code' && key !== 'id' && typeof config[key] === 'object' && config[key].min !== undefined
        );
       // 渲染出各个设备不同的配置参数
        if (hasParameters) {
            html += `
                <div class="config-section">
                    <div class="config-section-title">参数范围配置</div>
            `;

            Object.entries(config).forEach(([key, value]) => {
                if (key !== 'code' && key !== 'id' && typeof value === 'object' && value.min !== undefined) {
                    html += `
                        <div class="config-group">
                            <label>${this.getParameterDisplayName(key)}</label>
                            <div class="config-inputs">
                                <div class="input-group">
                                    <label>最小值</label>
                                    <input type="number" id="min_${key}" placeholder="最小值" value="${value.min}" step="any">
                                </div>
                                <div class="input-group">
                                    <label>最大值</label>
                                    <input type="number" id="max_${key}" placeholder="最大值" value="${value.max}" step="any">
                                </div>
                            </div>
                        </div>
                    `;
                }
            });

            html += '</div>';
        }

        html += '</div>';
        return html;
    }
```

当进行设备操作、SocketIO发送数据时，均向日志框中添加一条日志节点，通过class属性区分是发送日志还是接收日志。当设置自动滚动时，添加日志后还需要操作DOM元素向下滚动一定距离。当点击清空日志时，清空日志框中的所有日志节点。当进行日志筛选时，通过class属性设置部分日志的显示状态。

添加日志代码参考如下：

```typescript
    addLogEntry(type, message, timestamp = new Date().toISOString()) {
        //获取日志框
        const logContainer = document.getElementById('logContent');
        // 添加一个日志节点
        const logEntry = document.createElement('div');
        // 设置日志class样式
        logEntry.className = `log-entry ${type} show`;
        // 设置日志节点显示内容
        const time = new Date(timestamp).toLocaleTimeString();
        logEntry.innerHTML = `
            <span class="timestamp">[${time}]</span>
            <span class="message">${message}</span>
        `;
        // 在日志框中添加日志节点
        logContainer.appendChild(logEntry);
        // 如果设置自动滚动，则滚动到最底部
        if (this.autoScroll) {
            logContainer.scrollTop = logContainer.scrollHeight;
        }

        this.filterLogs(); // 应用当前筛选
    }
```

```kotlin
function getSVG(deviceId){
    if(deviceId==1||deviceId==2)return "td"
    else if(deviceId==3||deviceId==4)return "sjj"
    else if(deviceId==5)return "gzm"
    else if(deviceId==6)return "sjk"
    else if(deviceId==7)return "hj"
    else return "pl"
}
class TCPSimulator {
    constructor() {
        this.socket = io();
        this.autoScroll = true;
        this.logFilter = 'all';
        this.deviceConfig = {};
        this.currentEditingDevice = null;
        this.pendingDisconnectGroup = null;
        this.init();
    }

    async init() {
        await this.loadDeviceConfig();
        this.bindEvents();
        this.setupSocketListeners();
    }

    async loadDeviceConfig() {
        try {
            const response = await fetch('/api/config');
            this.deviceConfig = await response.json();
        } catch (error) {
            console.error('加载设备配置失败:', error);
        }
    }

    bindEvents() {
        // 清空日志
        document.getElementById('clearLog').addEventListener('click', () => {
            document.getElementById('logContent').innerHTML = '';
        });

        // 切换自动滚动
        document.getElementById('toggleAutoScroll').addEventListener('click', (e) => {
            this.autoScroll = !this.autoScroll;
            e.target.textContent = `自动滚动: ${this.autoScroll ? '开启' : '关闭'}`;
        });

        // 日志筛选
        document.getElementById('logFilter').addEventListener('change', (e) => {
            this.logFilter = e.target.value;
            this.filterLogs();
        });

        // 模态框关闭
        document.querySelectorAll('.close').forEach(closeBtn => {
            closeBtn.addEventListener('click', () => {
                this.closeModals();
            });
        });

        // 配置保存
        document.getElementById('saveConfig').addEventListener('click', () => {
            this.saveDeviceConfig();
        });

        // 配置取消
        document.getElementById('cancelConfig').addEventListener('click', () => {
            this.closeModals();
        });

        // 确认断开连接
        document.getElementById('confirmDisconnect').addEventListener('click', () => {
            this.confirmDisconnect();
        });

        // 取消断开连接
        document.getElementById('cancelDisconnect').addEventListener('click', () => {
            this.closeModals();
        });

        // 点击模态框外部关闭
        window.addEventListener('click', (e) => {
            const modals = document.querySelectorAll('.modal');
            modals.forEach(modal => {
                if (e.target === modal) {
                    this.closeModals();
                }
            });
        });
    }

    setupSocketListeners() {
        // 接收初始状态
        this.socket.on('initial_status', (data) => {
            console.log('初始状态:', data);
            this.updateConnectionStatus(data.connectionStatus);
            this.createDeviceButtons(data.deviceStatus);
        });

        // 连接状态更新
        this.socket.on('connection_status', (data) => {
            this.updateConnectionStatus({ [data.group]: data.connected });
        });

        // 设备状态更新
        this.socket.on('device_status_changed', (data) => {
            this.updateDeviceButton(data.deviceId, data.running);
        });

        // 数据发送日志
        this.socket.on('data_sent', (data) => {
            this.addLogEntry('sent', `设备 ${data.deviceId} 发送数据: ${JSON.stringify(data.data)}`, data.timestamp);
        });

        // 服务器响应
        this.socket.on('server_response', (data) => {
            this.addLogEntry('received', `服务器响应 (${this.getGroupDisplayName(data.group)}): ${data.message}`);
        });

        // 错误处理
        this.socket.on('error', (data) => {
            this.addLogEntry('error', data.message);
        });
    }

    updateConnectionStatus(status) {
        //获取连接状态框的DOM对象
        const container = document.getElementById('connectionStatus');
        //根据连接状态数据创建节点
        Object.entries(status).forEach(([group, connected]) => {
            const groupName = this.getGroupDisplayName(group);
            const elementId = `connection-${group}`;
            // 创建节点
            let element = document.getElementById(elementId);
            if (!element) {
                element = document.createElement('div');
                element.id = elementId;
                element.className = 'connection-item';
                container.appendChild(element);
            }
            // 根据连接状态修改样式
            element.className = `connection-item ${connected ? 'connected' : 'disconnected'}`;
            // 如果是已连接，需要添加断开连接按钮
            element.innerHTML = `
                <div>${groupName}</div>
                <img src="img/${connected?"connect.svg":"disconnect.svg"}"  class="disconnect-icon">
                ${connected ? `<button class="disconnect-btn" data-group="${group}">断开连接</button>` : ''}
            `;

            // 绑定断开连接事件
            if (connected) {
                element.querySelector('.disconnect-btn').addEventListener('click', (e) => {
                    e.stopPropagation();
                    this.showDisconnectConfirm(group, groupName);
                });
            }
        });
    }

    createDeviceButtons(deviceStatus) {
        // 获取设备操作按钮区域
        const container = document.getElementById('devicesGrid');
        container.innerHTML = '';
        // 循环渲染设备按钮
        Object.entries(deviceStatus).forEach(([deviceId, status]) => {
            const deviceItem = document.createElement('div');
            deviceItem.className = 'device-item';
            // 创建按钮
            const button = document.createElement('button');
            // 设置按钮样式
            button.className = `device-button ${status.running ? 'running' : 'stopped'}`;
            button.innerHTML = `
                <img src="img/${getSVG(deviceId)}${status.running ? '_connect' : '_disconnect'}.svg"/>
                <div class="device-button-inner">
                    <div>${deviceId}. ${status.type}</div>
                    <div class="device-status">${this.getDeviceStatusText(parseInt(deviceId), status.running)}</div>
                </div>
            `;
            // 设置点击事件，当点击时向服务端发送操作指令
            button.addEventListener('click', () => {
                this.toggleDevice(parseInt(deviceId));
            });
            // 设置编辑按钮
            const editButton = document.createElement('button');
            editButton.className = 'edit-btn';
            editButton.innerHTML = '编辑';
            editButton.addEventListener('click', () => {
                this.showConfigModal(parseInt(deviceId), status.type);
            });
            // 添加按钮到设备项
            deviceItem.appendChild(button);
            deviceItem.appendChild(editButton);
            container.appendChild(deviceItem);
        });
    }

    updateDeviceButton(deviceId, running) {
        const buttons = document.querySelectorAll('.device-button');
        buttons.forEach(button => {
            if (button.textContent.includes(`${deviceId}.`)) {
                button.className = `device-button ${running ? 'running' : 'stopped'}`;
                button.querySelector('img').src = `img/${getSVG(deviceId)}${running ? '_connect' : '_disconnect'}.svg`;
                const statusElement = button.querySelector('.device-status');
                if (statusElement) {
                    statusElement.textContent = this.getDeviceStatusText(deviceId, running);
                }
            }
        });
    }

    getDeviceStatusText(deviceId, running) {
        if (deviceId === 2 || deviceId === 4) {
            return running ? `运行中，点击停止` : `已停止，点击启动`;
        } else if (deviceId === 8) {
            return running ? `已连接，点击断开` : `未连接，点击连接`;
        } else {
            return running ? `数据已发送` : `点击发送数据`;
        }
    }

    getGroupDisplayName(groupKey) {
        const groupNames = {
            'tower_crane': '塔吊设备',
            'elevator': '升降机设备',
            'high_formwork': '高支模设备',
            'deep_foundation': '深基坑设备',
            'environment': '环境监测设备',
            'sprinkler': '喷淋设备'
        };
        return groupNames[groupKey] || groupKey;
    }

    toggleDevice(deviceId) {
        this.socket.emit('toggle_device', { deviceId });
        this.addLogEntry('info', `操作设备 ${deviceId}`);
    }

    async saveDeviceConfig() {
        if (!this.currentEditingDevice) return;

        const configKey = this.getConfigKey(this.currentEditingDevice);
        if (!configKey) return;

        const config = { ...this.deviceConfig[configKey] };
        let hasChanges = false;

        // 处理code和id
        const codeInput = document.getElementById('config_code');
        const idInput = document.getElementById('config_id');

        if (codeInput && codeInput.value !== config.code) {
            config.code = codeInput.value;
            hasChanges = true;
        }

        if (idInput) {
            // 处理id可能是数字或字符串
            const newId = isNaN(idInput.value) ? idInput.value : Number(idInput.value);
            if (newId !== config.id) {
                config.id = newId;
                hasChanges = true;
            }
        }

        // 处理其他参数
        Object.keys(config).forEach(key => {
            if (key !== 'code' && key !== 'id' && typeof config[key] === 'object') {
                const minInput = document.getElementById(`min_${key}`);
                const maxInput = document.getElementById(`max_${key}`);

                if (minInput && maxInput) {
                    const newMin = parseFloat(minInput.value);
                    const newMax = parseFloat(maxInput.value);

                    if (!isNaN(newMin) && !isNaN(newMax) && 
                        (newMin !== config[key].min || newMax !== config[key].max)) {
                        config[key] = { min: newMin, max: newMax };
                        hasChanges = true;
                    }
                }
            }
        });

        if (hasChanges) {
            this.deviceConfig[configKey] = config;

            try {
                const response = await fetch('/api/config', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify(this.deviceConfig),
                });

                const result = await response.json();
                if (result.success) {
                    this.addLogEntry('success', `设备 ${this.currentEditingDevice} 配置已保存`);
                    this.closeModals();
                } else {
                    this.addLogEntry('error', '保存配置失败');
                }
            } catch (error) {
                this.addLogEntry('error', '保存配置时发生错误');
            }
        } else {
            this.closeModals();
        }
    }

    closeModals() {
        document.querySelectorAll('.modal').forEach(modal => {
            modal.style.display = 'none';
        });
        this.currentEditingDevice = null;
        this.pendingDisconnectGroup = null;
    }

    addLogEntry(type, message, timestamp = new Date().toISOString()) {
        //获取日志框
        const logContainer = document.getElementById('logContent');
        // 添加一个日志节点
        const logEntry = document.createElement('div');
        // 设置日志class样式
        logEntry.className = `log-entry ${type} show`;
        // 设置日志节点显示内容
        const time = new Date(timestamp).toLocaleTimeString();
        logEntry.innerHTML = `
            <span class="timestamp">[${time}]</span>
            <span class="message">${message}</span>
        `;
        // 在日志框中添加日志节点
        logContainer.appendChild(logEntry);
        // 如果设置自动滚动，则滚动到最底部
        if (this.autoScroll) {
            logContainer.scrollTop = logContainer.scrollHeight;
        }

        this.filterLogs(); // 应用当前筛选
    }

    filterLogs() {
        const logEntries = document.querySelectorAll('.log-entry');
        logEntries.forEach(entry => {
            const isSent = entry.classList.contains('sent');
            const isReceived = entry.classList.contains('received');
            const isError = entry.classList.contains('error');
            const isInfo = entry.classList.contains('info');

            let shouldShow = false;

            switch (this.logFilter) {
                case 'all':
                    shouldShow = true;
                    break;
                case 'sent':
                    shouldShow = isSent || isInfo;
                    break;
                case 'received':
                    shouldShow = isReceived || isError;
                    break;
            }

            if (shouldShow) {
                entry.classList.add('show');
            } else {
                entry.classList.remove('show');
            }
        });
    }
```
