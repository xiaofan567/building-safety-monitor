任务描述

设备上下线记录功能用于实时监控设备的在线/离线状态变化，并提供历史记录查询。该页面采用响应式布局，包含设备筛选、数据表格展示和分页功能，帮助用户快速了解设备运行状态。

**重点**：API数据交互、分页组件、多条件筛选

**难点**：多条件筛选联动、分页组件封装、API数据交互

<script setup>
import { ref, computed, reactive, onMounted } from 'vue'
import { selectOnOffLineData } from '@/api/equipment'

// 筛选条件
const filters = ref({
  projectName: '',
  equipmentName: '',
  equipmentId: '',
  startDate: '',
  endDate: ''
})

// 分页相关数据
const pagination = reactive({
  currentPage: 1,
  pageSize: 10,
  total: 0,
  totalPages: 0,
})

// 设备数据
const equipmentData = ref([])

// 筛选后的设备数据
const filteredEquipment = computed(() => {
  return equipmentData.value
})

// 分页计算
const totalItems = computed(() => pagination.total)
const visiblePages = computed(() => {
  const pages = []
  const start = Math.max(1, pagination.currentPage - 2)
  const end = Math.min(pagination.totalPages, pagination.currentPage + 2)
  
  for (let i = start; i <= end; i++) {
    pages.push(i)
  }
  
  return pages
})

// 加载设备数据
const loadEquipmentData = async () => {
  try {
    const requestData = {
      page: pagination.currentPage,
      limit: pagination.pageSize,
    };
    
    // 添加筛选条件
    if (filters.value.equipmentName) {
      requestData.equipmentName = filters.value.equipmentName;
    }
    if (filters.value.equipmentId) {
      requestData.equipmentId = filters.value.equipmentId;
    }
    if (filters.value.startDate) {
      requestData.startTime = filters.value.startDate+" 00:00:00";;
    }
    if (filters.value.endDate) {
      requestData.endTime = filters.value.endDate+" 23:59:59";
    }

    const response = await selectOnOffLineData(requestData);
    console.log("设备上下线数据加载成功:", response);

    // 根据API响应结构处理数据
    if (response && response.records) {
      // 转换API数据格式为前端需要的格式
      equipmentData.value = response.records.map(item => ({
        id: item.id,
        equipmentName: item.equipmentName || '',
        equipmentCode: item.equipmentId || '',
        equipmentType: item.typeName || '',
        monitorTime: item.monitorTime || '',
        status: item.status === '1' ? '在线' : '离线'
      }));
      pagination.total = response.total || 0;
      pagination.totalPages = Math.ceil(pagination.total / pagination.pageSize);
    } else if (Array.isArray(response)) {
      equipmentData.value = response;
      pagination.total = response.length;
      pagination.totalPages = Math.ceil(pagination.total / pagination.pageSize);
    } else {
      console.warn("API响应格式异常:", response);
      equipmentData.value = [];
      pagination.total = 0;
      pagination.totalPages = 0;
    }
  } catch (error) {
    console.error("加载设备上下线数据失败:", error);
    if (error.response && error.response.status === 401) {
      console.error("认证失败，请重新登录");
      alert("认证失败，请重新登录");
    }
    equipmentData.value = [];
    pagination.total = 0;
    pagination.totalPages = 0;
  }
}

// 搜索功能
const searchAlarms = () => {
  pagination.currentPage = 1;
  loadEquipmentData();
}

// 重置筛选条件
const resetFilters = () => {
  filters.value = {
    equipmentName: '',
    equipmentId: '',
    startDate: '',
    endDate: ''
  }
  pagination.currentPage = 1;
  loadEquipmentData();
}

// 分页方法
const goToPage = (page) => {
  if (page < 1 || page > pagination.totalPages) return;
  pagination.currentPage = page;
  loadEquipmentData();
}

const prevPage = () => {
  if (pagination.currentPage > 1) {
    pagination.currentPage--;
    loadEquipmentData();
  }
}

const nextPage = () => {
  if (pagination.currentPage < pagination.totalPages) {
    pagination.currentPage++;
    loadEquipmentData();
  }
}

// 初始化加载数据
onMounted(() => {
  loadEquipmentData();
})
</script>

export async function selectOnOffLineData(requestData) {
  try {

        'Content-Type': 'application/json;charset=UTF-8'
      },
    });
    console.log("请求成功", response.data);
    return response.data;
  } catch (error) {
    console.error("请求失败", error);
    throw error;
  }
};
