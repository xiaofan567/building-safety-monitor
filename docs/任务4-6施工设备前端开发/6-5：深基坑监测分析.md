功能说明：

关键参数解释：

camera.position.set(30, 30, 30): 相机从右上前方较高位置看向场景（俯视视角）

camera.lookAt(0, 0, 0): 焦点对准场景原点

**7. 光源系统配置**

功能说明：
添加环境光和方向光，照亮3D场景，增强模型立体感和视觉效果。

代码实现：

```csharp
javascript
  // 7.1 添加光源系统
  addLights();
}

// 光源系统配置函数
function addLights() {
  // 7.2 清除现有光源（避免重复添加）
  scene.children = scene.children.filter(child => 
    !(child instanceof THREE.Light)
  );
  
  // 7.3 环境光：提供基础均匀照明
  const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
  scene.add(ambientLight);
  
  // 7.4 主方向光1：从右上前方照射，产生阴影和明暗对比
  const directionalLight1 = new THREE.DirectionalLight(0xffffff, 0.9);
  directionalLight1.position.set(30, 40, 20);
  scene.add(directionalLight1);
  
  // 7.5 辅助方向光2：从左下后方照射，补充暗部细节
  const directionalLight2 = new THREE.DirectionalLight(0xffffff, 0.5);
  directionalLight2.position.set(-20, 20, -10);
  scene.add(directionalLight2);
}
```

关键参数解释：

THREE.AmbientLight(): 环境光，均匀照亮所有物体

颜色0xffffff：白色光

强度0.6：中等偏亮亮度

THREE.DirectionalLight(): 方向光，模拟太阳光，有方向性

主光强度0.9：较强的主光源

位置(30, 40, 20)：从右上前方较高位置照射

辅助光强度0.5：中等强度的补充光源

位置(-20, 20, -10)：从左下后方照射

多光源设计：环境光 + 主方向光 + 辅助方向光，营造立体感

**10. 网格几何体创建函数**

功能说明：
辅助函数，用于创建网格墙的线几何体，生成水平和垂直网格线。

代码实现：

```gml
javascript
// 9.1 创建网格几何体辅助函数 
function createGridGeometry(width, height, gridSize, thickness) { 
 const geometry = new THREE.BufferGeometry();  const vertices = [];    
 // 9.2 横向网格线（水平线） 
  for (let y = 0; y <= height; y += gridSize) { 
     // 每条线由两个顶点组成：起点和终点
      vertices.push(-width/2 + thickness, y - height/2, 0); // 起点
       vertices.push(width/2 - thickness, y - height/2, 0);  // 终点  }    
     // 9.3 纵向网格线（垂直线）  
     for (let x = 0; x <= width; x += gridSize) {   
      // 每条线由两个顶点组成：起点和终点    
      vertices.push(x - width/2, -height/2 + thickness, 0);  // 起点   
       vertices.push(x - width/2, height/2 - thickness, 0);  // 终点  }    
       // 9.4 设置顶点数据 
        geometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));  
        return geometry; }
```

关键参数解释：

顶点生成原理：

横向线：从左侧到右侧，Y坐标递增

纵向线：从底部到顶部，X坐标递增

每条线：起点 + 终点 = 2个顶点

坐标计算：

X坐标范围：-width/2 到 width/2

Y坐标范围：-height/2 到 height/2

thickness：向内偏移，避免线与边界重叠

BufferGeometry优势：

高性能：直接操作缓冲区数据

低内存：紧凑的数据结构

灵活：可自定义顶点属性

// 加载深基坑数据
async function loadDeepExcavationData(assetId,index) {
  try {
    console.log('开始加载深基坑数据，assetId:', assetId);
    const response = await hfAndDData(assetId);
    console.log('API响应:', response);
    const monitorData = response.data;
    console.log('监测数据:', monitorData);
    
    if (monitorData) {
      // 更新实时监测数据 - 显示所有监测项的最新数据
      const realTimeData = [];
      
      // 遍历所有监测项类型
      Object.keys(monitorData).forEach(monitorType => {
        const items = monitorData[monitorType];
        console.log(`监测项 ${monitorType}:`, items);
        
        if (items && items.length > 0) {
          // 取每个监测项的最新一条数据
          const latestItem = items[0];
          console.log(`最新数据 ${monitorType}:`, latestItem);
          
          realTimeData.push({
            id: latestItem.id,
            name: getMonitorTypeName(monitorType),
            value: parseFloat(latestItem.monitorData),
            unit: latestItem.unit,
            time: latestItem.monitorTime,
            warningThreshold: getWarningThreshold(monitorType),
            alarmThreshold: getAlarmThreshold(monitorType)
          });

        // 给foundationConfigs.value的第index条数据的statusData的monitorPoints添加一条数据
        if (foundationConfigs.value[index] && foundationConfigs.value[index].statusData) {
          // 解析位置数据，将逗号分隔的字符串转换为数组
          const positionArray = latestItem.location ? 
            latestItem.location.split(',').map(coord => parseFloat(coord.trim())) : 
            [0, 0, 0];
          
          // 创建监测点数据
          const monitorPoint = {
            name: latestItem.monitorPointName || getMonitorTypeName(monitorType),
            value: parseFloat(latestItem.monitorData),
            unit: latestItem.unit || '',
            position: positionArray
          };
          
          // 添加到监测点数组
          foundationConfigs.value[index].statusData.monitorPoints.push(monitorPoint);
          
          console.log(`为深基坑#${index}添加监测点:`, monitorPoint);
        }

        }
      });
      
      console.log('实时监测数据:', realTimeData);
      currentMonitorData.value = realTimeData;
      
      // 处理沉降数据 - 取沉降监测项的前10条数据
      if (monitorData['沉降值'] && monitorData['沉降值'].length > 0) {
        const settlementItems = monitorData['沉降值'].slice(0, 10);
        console.log('沉降数据:', settlementItems);
        settlementData.value = settlementItems.map(item => [
          new Date(item.monitorTime).getTime(),
          parseFloat(item.monitorData) // 保持原始单位
        ]);
      } else {
        console.log('没有沉降数据');
        settlementData.value = [];
      }
    } else {
      console.log('监测数据为空');
      currentMonitorData.value = [];
      settlementData.value = [];
    }
    
    // 加载报警数据
    await loadAlarmData(assetId);
  } catch (error) {
    console.error("加载深基坑数据失败:", error);
    currentMonitorData.value = [];
    settlementData.value = [];
  }
}

// 生成最近7天的日期（包括今天）
function generateLast7Days() {
  const dates = [];
  const today = new Date();
  
  for (let i = 6; i >= 0; i--) {
    const date = new Date(today);
    date.setDate(today.getDate() - i);
    const dateString = date.toISOString().split('T')[0];
    dates.push(dateString);
  }
  
  return dates;
}

  camera.lookAt(0, 0, 0);
  
  // 创建渲染器
  renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setSize(
    modelContainer.value.clientWidth, 
    modelContainer.value.clientHeight
  );
  // 给渲染器DOM元素添加z-index，确保在标签下方但能接收事件
  renderer.domElement.style.zIndex = '1';
  modelContainer.value.appendChild(renderer.domElement);
  
  // 创建2D标签渲染器
  labelRenderer = new CSS2DRenderer();
  labelRenderer.setSize(
    modelContainer.value.clientWidth, 
    modelContainer.value.clientHeight
  );
  labelRenderer.domElement.style.position = 'absolute';
  labelRenderer.domElement.style.top = '0';
  labelRenderer.domElement.style.left = '0';
  labelRenderer.domElement.style.pointerEvents = 'none'; // 允许鼠标穿透标签
  labelRenderer.domElement.style.zIndex = '2'; // 标签在渲染器上方
  modelContainer.value.appendChild(labelRenderer.domElement);
  
  // 修复核心：轨道控制器绑定到WebGL渲染器的DOM元素（确保能接收鼠标事件）
  controls = new OrbitControls(camera, renderer.domElement);
  controls.enableDamping = true;
  controls.dampingFactor = 0.1;
  // 显式开启控制器功能（默认开启，但确保没有被关闭）
  controls.enableRotate = true; // 允许旋转
  controls.enableZoom = true;   // 允许缩放
  controls.enablePan = true;    // 允许平移
  
  // 创建基坑模型组
  foundationGroup = new THREE.Group();
  scene.add(foundationGroup);
  
  // 添加光源
  addLights();
  
  // 初始构建模型
  buildFoundationModel((foundationConfigs.value)[activeTab.value]);
  
  // 开始动画循环
  animate();
}

// 构建深基坑模型（网格墙+支撑）
function buildFoundationModel(config) {
  // 清除现有模型和标签
  while (foundationGroup.children.length > 0) {
    const child = foundationGroup.children[0];
    foundationGroup.remove(child);
    disposeObject(child);
  }
  if (labelRenderer && labelRenderer.domElement) {
    while (labelRenderer.domElement.firstChild) {
      labelRenderer.domElement.removeChild(labelRenderer.domElement.firstChild);
    }
  }
  
  // 解构配置参数
  const { 
    length, width, depth, supportLayers, supportSpacing, 
    gridSize, gridLineWidth, statusData 
  } = config;
  const wallThickness = 0.1;
  
  // 网格墙材质
  const normalGridMaterial = new THREE.LineBasicMaterial({
    color: 0x0A3D91, // 深蓝色网格
    linewidth: gridLineWidth
  });
  const warningGridMaterial = new THREE.LineBasicMaterial({
    color: 0xFFD700,
    linewidth: gridLineWidth + 1
  });
  const dangerGridMaterial = new THREE.LineBasicMaterial({
    color: 0xFF4500,
    linewidth: gridLineWidth + 2
  });
  
  // 支撑与底部材质
  const supportMaterial = new THREE.MeshStandardMaterial({ 
    color: 0x7B1FA2 
  });
  const bottomMaterial = new THREE.MeshStandardMaterial({ 
    color: 0x696969 
  });
  
  // 1. 绘制四周网格墙
  const wallHeight = depth;
  const walls = [
    { width: length, height: wallHeight, position: [0, -wallHeight/2, width/2], rotation: [0, 0, 0] },
    { width: length, height: wallHeight, position: [0, -wallHeight/2, -width/2], rotation: [0, 0, 0] },
    { width: width, height: wallHeight, position: [-length/2, -wallHeight/2, 0], rotation: [0, Math.PI/2, 0] },
    { width: width, height: wallHeight, position: [length/2, -wallHeight/2, 0], rotation: [0, Math.PI/2, 0] }
  ];
  
  walls.forEach((wall, index) => {
    let material = normalGridMaterial;
    if (statusData.dangerWalls.includes(index)) material = dangerGridMaterial;
    else if (statusData.warningWalls.includes(index)) material = warningGridMaterial;
    
    const gridGeometry = createGridGeometry(wall.width, wall.height, gridSize, wallThickness);
    const gridMesh = new THREE.LineSegments(gridGeometry, material);
    gridMesh.position.set(...wall.position);
    gridMesh.rotation.set(...wall.rotation);
    foundationGroup.add(gridMesh);
  });
  
  // 2. 基坑底部
  const bottomGeometry = new THREE.BoxGeometry(length - 0.5, 0.3, width - 0.5);
  const bottomMesh = new THREE.Mesh(bottomGeometry, bottomMaterial);
  bottomMesh.position.set(0, -depth, 0);
  foundationGroup.add(bottomMesh);
  
  // 3. 格栅支撑
  const supportSpacingY = depth / (supportLayers + 1);
  for (let i = 0; i < supportLayers; i++) {
    const supportY = -depth + supportSpacingY * (i + 1);
    let material = supportMaterial;
    if (statusData.dangerSupports.includes(i)) material = new THREE.MeshStandardMaterial({ color: 0xFF4500 });
    else if (statusData.warningSupports.includes(i)) material = new THREE.MeshStandardMaterial({ color: 0xFFD700 });
    
    // 横向格栅
    for (let x = -length/2 + 0.5; x < length/2; x += supportSpacing) {
      const xSupport = new THREE.Mesh(new THREE.BoxGeometry(0.2, 0.2, width - 1), material);
      xSupport.position.set(x, supportY, 0);
      foundationGroup.add(xSupport);
    }
    
    // 纵向格栅
    for (let z = -width/2 + 0.5; z < width/2; z += supportSpacing) {
      const zSupport = new THREE.Mesh(new THREE.BoxGeometry(length - 1, 0.2, 0.2), material);
      zSupport.position.set(0, supportY, z);
      foundationGroup.add(zSupport);
    }
  }
  
  // 4. 周边地面
  const groundMesh = new THREE.Mesh(
    new THREE.PlaneGeometry(length + 4, width + 4),
    new THREE.MeshStandardMaterial({ color: 0x8FBC8F, transparent: true, opacity: 0.5, side: THREE.DoubleSide })
  );
  groundMesh.rotation.x = -Math.PI / 2;
  groundMesh.position.set(0, 0.1, 0);
  foundationGroup.add(groundMesh);
  
  // 5. 数据标签
  statusData.monitorPoints.forEach(point => {
    const labelDiv = document.createElement('div');
    labelDiv.className = 'data-label';
    labelDiv.innerHTML = `<div style='font-weight:600'>${point.name}</div><div>${point.value} ${point.unit}</div>`;
    const label = new CSS2DObject(labelDiv);
    label.position.set(...point.position);
    foundationGroup.add(label);
  });
}

// 创建网格几何体
function createGridGeometry(width, height, gridSize, thickness) {
  const geometry = new THREE.BufferGeometry();
  const vertices = [];
  
  // 横向网格线
  for (let y = 0; y <= height; y += gridSize) {
    vertices.push(-width/2 + thickness, y - height/2, 0);
    vertices.push(width/2 - thickness, y - height/2, 0);
  }
  
  // 纵向网格线
  for (let x = 0; x <= width; x += gridSize) {
    vertices.push(x - width/2, -height/2 + thickness, 0);
    vertices.push(x - width/2, height/2 - thickness, 0);
  }
  
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));
  return geometry;
}

// 添加光源
function addLights() {
  scene.children = scene.children.filter(child => !(child instanceof THREE.Light));
  const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
  scene.add(ambientLight);
  const directionalLight1 = new THREE.DirectionalLight(0xffffff, 0.9);
  directionalLight1.position.set(30, 40, 20);
  scene.add(directionalLight1);
  const directionalLight2 = new THREE.DirectionalLight(0xffffff, 0.5);
  directionalLight2.position.set(-20, 20, -10);
  scene.add(directionalLight2);
}

// 工具函数
function disposeObject(obj) {
  if (obj.geometry) obj.geometry.dispose();
  if (obj.material) {
    Array.isArray(obj.material) ? obj.material.forEach(m => m.dispose()) : obj.material.dispose();
  }
}

  renderer.render(scene, camera);
  labelRenderer.render(scene, camera);
}
